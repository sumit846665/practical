#include 

int main() {
    char data[50], key[20];
    int i, j;

    printf("Enter Data: ");
    scanf("%s", data);

    printf("Enter Key: ");
    scanf("%s", key);

    int data_len = strlen(data);
    int key_len = strlen(key);

    for(i = data_len; i < data_len + key_len - 1; i++)
        data[i] = '0';

    data[i] = '\0';

    char temp[50];
    strcpy(temp, data);

    for(i = 0; i < data_len; i++) {
        if(temp[i] == '1') {
            for(j = 0; j < key_len; j++) {
                temp[i + j] = (temp[i + j] == key[j]) ? '0' : '1';
            }
        }
    }

    printf("Remainder: ");
    for(i = data_len; i < data_len + key_len - 1; i++)
        printf("%c", temp[i]);

    printf("\nFinal Data: ");
    for(i = 0; i < data_len; i++)
        printf("%c", data[i]);

    for(i = data_len; i < data_len + key_len - 1; i++)
        printf("%c", temp[i]);

    return 0;
}

2️⃣ IP Validation + Full Details

#include 

int main() {
    int a,b,c,d;

    printf("Enter IP (a.b.c.d): ");
    scanf("%d.%d.%d.%d",&a,&b,&c,&d);

    if(a<0||a>255||b<0||b>255||c<0||c>255||d<0||d>255){
        printf("Invalid IP\n");
        return 0;
    }

    printf("Valid IP\n");

    int mask[4] = {0,0,0,0};
    int hosts = 0;

    if(a>=0 && a<=127){
        printf("Class A\n");
        mask[0]=255; mask[1]=0; mask[2]=0; mask[3]=0;
        hosts = 16777214;
    }
    else if(a<=191){
        printf("Class B\n");
        mask[0]=255; mask[1]=255; mask[2]=0; mask[3]=0;
        hosts = 65534;
    }
    else if(a<=223){
        printf("Class C\n");
        mask[0]=255; mask[1]=255; mask[2]=255; mask[3]=0;
        hosts = 254;
    }
    else if(a<=239){
        printf("Class D (Multicast)\n");
    }
    else{
        printf("Class E (Reserved)\n");
    }

    printf("Default Mask: %d.%d.%d.%d\n",mask[0],mask[1],mask[2],mask[3]);

    printf("Network Address: %d.%d.%d.%d\n",
        a & mask[0],
        b & mask[1],
        c & mask[2],
        d & mask[3]);

    printf("Broadcast Address: %d.%d.%d.%d\n",
        (a & mask[0]) | (~mask[0] & 255),
        (b & mask[1]) | (~mask[1] & 255),
        (c & mask[2]) | (~mask[2] & 255),
        (d & mask[3]) | (~mask[3] & 255));

    if(hosts != 0)
        printf("Number of Hosts: %d\n", hosts);

    if((a==10) || (a==172 && b>=16 && b<=31) || (a==192 && b==168))
        printf("Type: Private\n");
    else
        printf("Type: Public\n");

    return 0;
}
